The code that draws it

// one fetch gives price and floor together
const s = await (await fetch(
  `${API}/v2/dex/summary?id=${pair}`)).json();
const price = Number(s.priceNative);
const floor = Number(s.floorPriceNative);

// hide the floor until the launch has one,
// and when every token has been burned
const hasFloor = floor > 0 &&
  Number(s.circulatingSupply) > 0;

priceLine.push({ time, value: price });
if (hasFloor) {
  floorLine.push({ time, value: floor });
}
